🎯 What is KEM?

KEM enables secure key exchange using public-key cryptography to establish a shared secret.

// Traditional: Diffie-Hellman key exchange // Complex, prone to errors // New: KEM API (simpler!) // 1. Sender encapsulates a shared secret // 2. Receiver decapsulates to get same secret // 3. Use shared secret for symmetric encryption // Benefits: // - Post-quantum ready (with right algorithms) // - Simpler API // - More secure defaults

🔑 Generate Key Pair

// Generate KEM key pair KeyPairGenerator kpg = KeyPairGenerator .getInstance("X25519"); KeyPair keyPair = kpg.generateKeyPair(); PublicKey publicKey = keyPair.getPublic(); PrivateKey privateKey = keyPair.getPrivate(); // Sender gets public key // Receiver keeps private key

📤 Encapsulation (Sender)

// Sender creates shared secret KEM kem = KEM.getInstance("DHKEM"); // Encapsulate using receiver's public key KEM.Encapsulator encapsulator = kem.newEncapsulator(receiverPublicKey); KEM.Encapsulated encapsulated = encapsulator.encapsulate(); // Get shared secret (use for encryption) SecretKey sharedSecret = encapsulated.key(); // Get encapsulation (send to receiver) byte[] encapsulation = encapsulated.encapsulation(); // Send encapsulation to receiver sendToReceiver(encapsulation); // Encrypt data with sharedSecret Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding"); cipher.init(Cipher.ENCRYPT_MODE, sharedSecret); byte[] encrypted = cipher.doFinal(data);

📥 Decapsulation (Receiver)

// Receiver extracts shared secret KEM kem = KEM.getInstance("DHKEM"); // Decapsulate using private key KEM.Decapsulator decapsulator = kem.newDecapsulator(receiverPrivateKey); // Receive encapsulation from sender byte[] encapsulation = receiveFromSender(); // Decapsulate to get shared secret SecretKey sharedSecret = decapsulator.decapsulate(encapsulation); // Now receiver has same sharedSecret! // Use it to decrypt data Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding"); cipher.init(Cipher.DECRYPT_MODE, sharedSecret, params); byte[] decrypted = cipher.doFinal(encrypted);

💼 Complete Example

class SecureMessaging { // Receiver setup static KeyPair receiverSetup() throws Exception { KeyPairGenerator kpg = KeyPairGenerator.getInstance("X25519"); return kpg.generateKeyPair(); } // Sender encrypts message static EncryptedMessage sendMessage( PublicKey receiverPublicKey, String message) throws Exception { KEM kem = KEM.getInstance("DHKEM"); KEM.Encapsulator enc = kem.newEncapsulator(receiverPublicKey); KEM.Encapsulated encapsulated = enc.encapsulate(); SecretKey key = encapsulated.key(); byte[] encapsulation = encapsulated.encapsulation(); Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding"); cipher.init(Cipher.ENCRYPT_MODE, key); byte[] ciphertext = cipher.doFinal( message.getBytes(StandardCharsets.UTF_8) ); byte[] iv = cipher.getIV(); return new EncryptedMessage( encapsulation, ciphertext, iv ); } // Receiver decrypts message static String receiveMessage( PrivateKey receiverPrivateKey, EncryptedMessage msg) throws Exception { KEM kem = KEM.getInstance("DHKEM"); KEM.Decapsulator dec = kem.newDecapsulator(receiverPrivateKey); SecretKey key = dec.decapsulate(msg.encapsulation()); Cipher cipher = Cipher.getInstance("AES/GCM/NoPadding"); GCMParameterSpec params = new GCMParameterSpec(128, msg.iv()); cipher.init(Cipher.DECRYPT_MODE, key, params); byte[] plaintext = cipher.doFinal(msg.ciphertext()); return new String(plaintext, StandardCharsets.UTF_8); } } record EncryptedMessage( byte[] encapsulation, byte[] ciphertext, byte[] iv ) {}
