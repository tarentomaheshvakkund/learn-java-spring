🎯 What is HexFormat?

A utility class for converting between byte arrays and hexadecimal string representations.

// Old way: Manual conversion (error-prone!) String toHex(byte[] bytes) { StringBuilder sb = new StringBuilder(); for (byte b : bytes) { sb.append(String.format("%02x", b)); } return sb.toString(); } // New way: HexFormat (built-in!) HexFormat hex = HexFormat.of(); byte[] bytes = {0x12, 0x34, 0x56, 0x78}; String hexString = hex.formatHex(bytes); // Output: "12345678"

📤 Bytes to Hex

HexFormat hex = HexFormat.of(); // Format byte array byte[] bytes = {(byte) 0xCA, (byte) 0xFE, (byte) 0xBA, (byte) 0xBE}; String result = hex.formatHex(bytes); // Output: "cafebabe" // Format single byte byte b = (byte) 0xFF; String single = hex.toHexDigits(b); // Output: "ff" // Format with uppercase HexFormat upper = HexFormat.ofDelimiter("").withUpperCase(); String upperHex = upper.formatHex(bytes); // Output: "CAFEBABE" // Format with delimiter HexFormat delim = HexFormat.ofDelimiter(":"); String formatted = delim.formatHex(bytes); // Output: "ca:fe:ba:be"

📥 Hex to Bytes

HexFormat hex = HexFormat.of(); // Parse hex string to bytes String hexString = "48656c6c6f"; byte[] bytes = hex.parseHex(hexString); System.out.println(new String(bytes)); // Output: "Hello" // Parse with delimiter HexFormat colonHex = HexFormat.ofDelimiter(":"); byte[] mac = colonHex.parseHex("00:11:22:33:44:55"); // Returns: {0x00, 0x11, 0x22, 0x33, 0x44, 0x55} // Parse with prefix/suffix HexFormat formatted = HexFormat.ofDelimiter(",") .withPrefix("0x"); byte[] data = formatted.parseHex("0x01,0x02,0x03"); // Returns: {0x01, 0x02, 0x03}

🎨 Custom Formatting

// Uppercase with colon delimiter HexFormat macFormat = HexFormat.ofDelimiter(":") .withUpperCase(); byte[] macAddress = {0x00, 0x1a, 0x2b, 0x3c, 0x4d, 0x5e}; String mac = macFormat.formatHex(macAddress); // Output: "00:1A:2B:3C:4D:5E" // With prefix "0x" and comma delimiter HexFormat arrayFormat = HexFormat.ofDelimiter(", ") .withPrefix("0x") .withUpperCase(); byte[] data = {0xde, (byte) 0xad, (byte) 0xbe, (byte) 0xef}; String formatted = arrayFormat.formatHex(data); // Output: "0xDE, 0xAD, 0xBE, 0xEF" // UUID format (with hyphens) HexFormat uuidFormat = HexFormat.ofDelimiter("") .withLowerCase(); byte[] uuid = new byte[16]; new Random().nextBytes(uuid); String uuidStr = String.format( "%s-%s-%s-%s-%s", uuidFormat.formatHex(uuid, 0, 4), uuidFormat.formatHex(uuid, 4, 6), uuidFormat.formatHex(uuid, 6, 8), uuidFormat.formatHex(uuid, 8, 10), uuidFormat.formatHex(uuid, 10, 16) ); // Output: "a1b2c3d4-e5f6-7890-abcd-ef1234567890"

💼 Practical Examples

// Hash to hex string MessageDigest md = MessageDigest.getInstance("SHA-256"); byte[] hash = md.digest("Hello".getBytes()); String hashHex = HexFormat.of().formatHex(hash); // Color codes byte[] rgb = {(byte) 0xFF, 0x5A, 0x3C}; String color = "#" + HexFormat.of() .withUpperCase() .formatHex(rgb); // Output: "#FF5A3C" // Network packet debugging void debugPacket(byte[] packet) { HexFormat hex = HexFormat.ofDelimiter(" ") .withUpperCase(); System.out.println("Packet: " + hex.formatHex(packet)); } // IP address (v6) byte[] ipv6 = {0x20, 0x01, 0x0d, (byte) 0xb8, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x01}; HexFormat ipFormat = HexFormat.ofDelimiter(":"); String ipv6Str = ipFormat.formatHex(ipv6); // Output: "2001:0db8:0000:0000:0000:0000:0000:0001"

⚡ Performance Tips

// ✅ Reuse HexFormat instances private static final HexFormat HEX = HexFormat.of(); public String toHex(byte[] bytes) { return HEX.formatHex(bytes); } // ✅ Use appropriate methods // For single byte: toHexDigits(byte) // For byte array: formatHex(byte[]) // For range: formatHex(byte[], int, int) // ✅ Choose lowercase (default) for performance HexFormat lower = HexFormat.of(); // Faster HexFormat upper = HexFormat.of().withUpperCase(); // Slower // ❌ Don't create new instances repeatedly for (byte[] data : datasets) { // Bad: Creates new instance each time String hex = HexFormat.of().formatHex(data); } // ✅ Create once, use many times HexFormat hex = HexFormat.of(); for (byte[] data : datasets) { String hexStr = hex.formatHex(data); }
